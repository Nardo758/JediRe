#!/usr/bin/env python3
"""
Fix 3: Repoint LayeredValue imports from renamed files to canonical types/layered-value.ts
"""

import os
import re

fixes = [
    ('src/services/coefficient-resolver.service.ts', 'traffic-calibration.types', '../types/layered-value'),
    ('src/services/deterministic/proforma-assumptions-bridge.ts', 'document-extraction/types', '../../types/layered-value'),
    ('src/services/inflation/index.ts', 'replacement-cost-v2.service', '../../types/layered-value'),
    ('src/services/proforma-seeder.service.ts', 'document-extraction/types', '../types/layered-value'),
    ('src/services/proforma/periodic-field.types.ts', 'document-extraction/types', '../../types/layered-value'),
    ('src/services/tax/compositeResolver.ts', './types', '../../types/layered-value'),
]

for filepath, source_substr, canonical_path in fixes:
    if not os.path.exists(filepath):
        print(f"SKIP: {filepath} not found")
        continue
    
    with open(filepath, 'r') as f:
        content = f.read()
    
    pattern = r"^import\s+(?:type\s+)?\{([^}]*)\}\s+from\s+(['\"][^'\"]*" + re.escape(source_substr) + r"[^'\"]*['\"]);?\s*$"
    matches = list(re.finditer(pattern, content, re.MULTILINE))
    
    if not matches:
        print(f"NO MATCH: {filepath}")
        continue
    
    for m in matches:
        imports = m.group(1)
        source = m.group(2)
        
        if 'LayeredValue' not in imports:
            continue
        
        items = [i.strip() for i in imports.split(',') if i.strip()]
        other_items = [i for i in items if 'LayeredValue' not in i]
        
        old_block = m.group(0)
        new_lines = []
        
        if other_items:
            new_lines.append(f"import {{ {', '.join(other_items)} }} from {source};")
        
        new_lines.append(f"import type {{ LayeredValue }} from '{canonical_path}';")
        
        content = content.replace(old_block, '\n'.join(new_lines), 1)
        print(f"FIXED: {filepath}")
        for line in new_lines:
            print(f"  -> {line}")
        print()
    
    with open(filepath, 'w') as f:
        f.write(content)

print("Done.")
PYEOF
