import { Pool } from 'pg';
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

function classifyDocType(filename: string): string {
  const lower = filename.toLowerCase();
  if (/\\brr\\b/.test(lower) || /rent\\s*roll/.test(lower)) return 'rent_roll';
  if (/\\binc\\b/.test(lower) || /\\bt12\\b/.test(lower) || /trailing\\s*12/.test(lower) || /income\\s*(stmt|statement)/.test(lower)) return 't12';
  if (/\\btax\\s*bill\\b/.test(lower) || /\\btaxable\\b/.test(lower) || /\\btax\\b.*\\b(201\\d|202\\d)/.test(lower)) return 'tax';
  if (/offering\\s*memorandum|offering_memorandum|\\bom\\b/.test(lower)) return 'om';
  if (/unit\\s*mix|unitmix/.test(lower)) return 'unit_mix';
  return 'unknown';
}

function extractYear(filename: string): number | null {
  // Pattern: (2019.05.31) or (2019.05)
  let m = filename.match(/\\((\\d{4})[._]\\d{2}(?:[._]\\d{2})?\\)/);
  if (m) { const y = parseInt(m[1]); if (y >= 1980 && y <= 2030) return y; }
  // Pattern: 080819 (MMDDYY)
  m = filename.match(/\\b(\\d{2})(\\d{2})(\\d{2})\\b/);
  if (m) { const yy = parseInt(m[3]); const y = yy < 50 ? 2000 + yy : 1900 + yy; if (y >= 1980 && y <= 2030) return y; }
  // Pattern: 2019.05.31 or 2019-05-31
  m = filename.match(/\\b(\\d{4})[._\\-]\\d{2}[._\\-]\\d{2}\\b/);
  if (m) { const y = parseInt(m[1]); if (y >= 1980 && y <= 2030) return y; }
  // Pattern: standalone year
  m = filename.match(/\\b(201\\d|202\\d)\\b/);
  if (m) { return parseInt(m[1]); }
  return null;
}

async function main() {
  const res = await pool.query(\"SELECT id, original_filename FROM data_library_files WHERE scope_id='GLOBAL' AND uploaded_at::date = '2026-05-21'\");
  const files = res.rows;
  
  const clusters = new Map<string, { files: Array<{docType: string; year: number|null}> }>();
  
  for (const f of files) {
    const base = f.original_filename.replace(/\\.[^.]+$/, '');
    const docType = classifyDocType(base);
    const year = extractYear(base);
    
    let cleaned = base
      .replace(/\\b(rent\\s*roll|rentroll|rr|t12|t-12|om|offering\\s*memorandum|tax\\s*bill|taxable|unit\\s*mix|financials|inc|income)\\b/gi, ' ')
      .replace(/\\(\\d{4}[._]\\d{2}(?:[._]\\d{2})?\\)/g, ' ')
      .replace(/\\d{4}[._\\-]\\d{2}[._\\-]\\d{2}/g, ' ')
      .replace(/\\d{6}/g, ' ')
      .replace(/\\b(techno|modified|copy\\s*of|summary|statement|detail|preliminary|historical|forecast|budget)\\b/gi, ' ')
      .replace(/\\s+/g, ' ').trim();
    
    const normKey = cleaned.toLowerCase().replace(/[^a-z0-9]/g, '') || 'unknown';
    if (!clusters.has(normKey)) clusters.set(normKey, { files: [] });
    clusters.get(normKey)!.files.push({ docType, year });
  }

  let omCount = 0, opCount = 0, bothCount = 0, temporalCount = 0;

  for (const [name, data] of clusters) {
    const types = new Set(data.files.map(f => f.docType));
    const hasOM = types.has('om');
    const hasOperating = types.has('rent_roll') || types.has('t12') || types.has('tax') || types.has('unit_mix');
    if (hasOM) omCount++;
    if (hasOperating) opCount++;
    if (hasOM && hasOperating) bothCount++;

    const years = new Set<number>();
    for (const f of data.files) {
      if (f.year && (f.docType === 'rent_roll' || f.docType === 't12')) years.add(f.year);

      const span = Math.max(...years) - Math.min(...years);
      if (span >= 2) {
        temporalCount++;
        console.log('TEMPORAL:', name, Array.from(years).sort().join(','), 'span=' + span);
      }
    }
  }

  console.log('\\n=== RECOMPUTED ===');
  console.log('Clusters:', clusters.size);
  console.log('OM only:', omCount);
  console.log('Operating only:', opCount);
  console.log('BOTH OM + operating:', bothCount);
  console.log('Temporal (≥2yr span):', temporalCount);
  
  await pool.end();
}
